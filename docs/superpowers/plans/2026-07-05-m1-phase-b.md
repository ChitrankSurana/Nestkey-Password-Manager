> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Tauri desktop shell + IPC command layer + React UI on top of the existing
`pwmgr-core` library, delivering M1's remaining scope: register/login, vault CRUD, trash,
real-time search, auto-lock (idle + manual), dark theme, and keyboard-accessible tooltips.

**Architecture:** Cargo workspace (`core` + `src-tauri`). `src-tauri` holds Tauri-managed state
(`Mutex`-guarded registry connection, vault connection, `Session`, config) behind `#[tauri::command]`
handlers that convert core types to serializable DTOs — `pwmgr-core` itself stays serde-free and
`VaultKey` never crosses the IPC boundary. A background tick drives `Session::check_expiry` and
emits a `vault-locked` event; the frontend calls a `touch` command on user activity.

**Tech Stack:** Rust (Tauri 2.x, rusqlite/SQLCipher via `pwmgr-core`), React 18 + TypeScript + Vite,
Tailwind CSS (dark theme only for M1).

```rust
use std::path::{Path, PathBuf};

pub struct AppPaths {
    root: PathBuf,
}

impl AppPaths {
    pub fn new(root: PathBuf) -> Self {
        AppPaths { root }
    }

    /// Resolves to `~/.pwmgr` on a real machine (via `dirs::home_dir()` at the call site in
    /// `main.rs`); tests construct this directly with a tempdir root instead.
    pub fn root(&self) -> &Path {
        &self.root
    }

    pub fn vaults_dir(&self) -> PathBuf {
        self.root.join("vaults")
    }

    pub fn backups_dir(&self) -> PathBuf {
        self.root.join("backups")
    }

    pub fn registry_db_path(&self) -> PathBuf {
        self.root.join("registry.db")
    }

    pub fn vault_path(&self, username: &str) -> PathBuf {
        self.vaults_dir().join(format!("{}.vault", username))
    }

    pub fn config_path(&self) -> PathBuf {
        self.root.join("config.json")
    }

    /// Creates root/vaults/backups if missing and restricts each to the current user only.
    /// THREAT_MODEL A2 (shared machine) lists OS-level file permissions as a mitigation that
    /// nothing in pwmgr-core sets — this is the layer that owns app-directory scaffolding, so it
    /// owns closing that gap too.
    pub fn ensure_app_dirs(&self) -> std::io::Result<()> {
        for dir in [self.root(), &self.vaults_dir(), &self.backups_dir()] {
            std::fs::create_dir_all(dir)?;
            Self::restrict_to_owner(dir)?;
        }
        Ok(())
    }

    #[cfg(unix)]
    fn restrict_to_owner(dir: &Path) -> std::io::Result<()> {
        use std::os::unix::fs::PermissionsExt;
        std::fs::set_permissions(dir, std::fs::Permissions::from_mode(0o700))
    }

    #[cfg(windows)]
    fn restrict_to_owner(dir: &Path) -> std::io::Result<()> {
        // Windows ACLs aren't exposed via std::fs::Permissions. icacls restricts the directory to
        // the current user, mirroring Unix's 0o700 (owner rwx, nobody else). Best-effort: if
        // icacls itself is unavailable (non-standard Windows install), don't fail app startup over
        // it - log and continue, since this is defense-in-depth, not the primary security boundary
        // (that's the SQLCipher/Argon2id encryption pwmgr-core already provides).
        let username = std::env::var("USERNAME").unwrap_or_default();
        let output = std::process::Command::new("icacls")
            .arg(dir)
            .arg("/inheritance:r")
            .arg("/grant:r")
            .arg(format!("{}:F", username))
            .output();
        match output {
            Ok(o) if o.status.success() => Ok(()),
            _ => {
                eprintln!("warning: could not restrict permissions on {:?}", dir);
                Ok(())
            }
        }
    }
}
```

```rust
use serde::{Deserialize, Serialize};
use std::path::Path;

const VALID_AUTO_LOCK_MINUTES: [u32; 3] = [1, 5, 15];

fn default_theme() -> String {
    "dark".to_string()
}

fn default_auto_lock_minutes() -> u32 {
    5
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub struct Config {
    #[serde(default = "default_theme")]
    pub theme: String,
    #[serde(default = "default_auto_lock_minutes")]
    pub auto_lock_minutes: u32,
}

impl Default for Config {
    fn default() -> Self {
        Config { theme: default_theme(), auto_lock_minutes: default_auto_lock_minutes() }
    }
}

impl Config {
    pub fn load(path: &Path) -> std::io::Result<Self> {
        if !path.exists() {
            return Ok(Config::default());
        }
        let raw = std::fs::read_to_string(path)?;
        // Per-field #[serde(default = ...)] means a config.json missing a field (e.g. saved by
        // an older app version before a new field existed) only loses that one field to its
        // default, not the whole struct - a parse error here means genuinely malformed content
        // (not valid JSON, or a field with the wrong type), which is rare enough to warrant a
        // diagnostic before falling back, same precedent as paths.rs's icacls best-effort warning.
        let parsed: Config = match serde_json::from_str(&raw) {
            Ok(config) => config,
            Err(e) => {
                eprintln!("warning: config.json unreadable ({e}), resetting to defaults");
                Config::default()
            }
        };
        if !VALID_AUTO_LOCK_MINUTES.contains(&parsed.auto_lock_minutes) {
            return Ok(Config { auto_lock_minutes: default_auto_lock_minutes(), ..parsed });
        }
        Ok(parsed)
    }

    pub fn save(&self, path: &Path) -> std::io::Result<()> {
        let raw = serde_json::to_string_pretty(self)?;
        std::fs::write(path, raw)
    }
}
```

```rust
use serde::Serialize;

#[derive(Debug, Serialize)]
pub struct CommandError {
    pub code: String,
    pub message: String,
}

impl CommandError {
    fn new(code: &str, message: &str) -> Self {
        CommandError { code: code.to_string(), message: message.to_string() }
    }
}

impl From<pwmgr_core::auth::AuthError> for CommandError {
    fn from(e: pwmgr_core::auth::AuthError) -> Self {
        use pwmgr_core::auth::AuthError;
        match e {
            // Single generic message on any auth failure - TRD §3.2 / THREAT_MODEL wrong-password
            // oracle: no distinguishing signal about which of username/password/vault was wrong.
            AuthError::InvalidCredentials => {
                CommandError::new("invalid_credentials", "Incorrect master password.")
            }
            AuthError::UsernameTaken => {
                CommandError::new("username_taken", "That username is already taken.")
            }
            AuthError::Registry(_) | AuthError::Vault(_) | AuthError::Crypto(_) => {
                CommandError::new("internal_error", "Something went wrong. Please try again.")
            }
        }
    }
}

impl From<pwmgr_core::vault::VaultError> for CommandError {
    fn from(e: pwmgr_core::vault::VaultError) -> Self {
        use pwmgr_core::vault::VaultError;
        match e {
            VaultError::NotFound => CommandError::new("not_found", "That item no longer exists."),
            VaultError::UnlockFailed | VaultError::Db(_) | VaultError::Crypto(_) => {
                CommandError::new("internal_error", "Something went wrong. Please try again.")
            }
        }
    }
}

/// Not backed by an unlocked session - distinct from any pwmgr-core error because pwmgr-core has
/// no concept of "session"; that state lives entirely in this Tauri layer's AppState.
pub fn locked_error() -> CommandError {
    CommandError::new("locked", "Vault is locked. Please log in again.")
}
```

```rust
// src-tauri/src/commands.rs (top of file, above the Task 9 unlock_status command)
use crate::dto::AuthSuccess;
use crate::error::CommandError;
use crate::paths::AppPaths;
use crate::state::AppState;
use pwmgr_core::session::SystemClock;
use tauri::State;

pub fn do_register(
    state: &AppState,
    paths: &AppPaths,
    username: &str,
    master_password: &str,
) -> Result<AuthSuccess, CommandError> {
    let vault_path = paths.vault_path(username);
    let key = {
        let registry = state.registry.lock().unwrap();
        pwmgr_core::auth::register_user(&registry, &vault_path, username, master_password)?
    };
    let conn = pwmgr_core::vault::open_vault(&vault_path, &key)?;
    state.session.lock().unwrap().unlock(key, &SystemClock);
    *state.vault.lock().unwrap() = Some(conn);
    *state.vault_path.lock().unwrap() = Some(vault_path);
    Ok(AuthSuccess { username: username.to_string() })
}

pub fn do_login(
    state: &AppState,
    paths: &AppPaths,
    username: &str,
    master_password: &str,
) -> Result<AuthSuccess, CommandError> {
    let vault_path = paths.vault_path(username);
    let key = {
        let registry = state.registry.lock().unwrap();
        pwmgr_core::auth::login(&registry, &vault_path, username, master_password)?
    };
    let conn = pwmgr_core::vault::open_vault(&vault_path, &key)?;
    state.session.lock().unwrap().unlock(key, &SystemClock);
    *state.vault.lock().unwrap() = Some(conn);
    *state.vault_path.lock().unwrap() = Some(vault_path);
    Ok(AuthSuccess { username: username.to_string() })
}

pub fn do_lock(state: &AppState) {
    state.session.lock().unwrap().lock();
    *state.vault.lock().unwrap() = None;
    *state.vault_path.lock().unwrap() = None;
}

pub fn do_touch(state: &AppState) {
    state.session.lock().unwrap().touch(&SystemClock);
}

#[tauri::command]
pub fn register_user(
    state: State<AppState>,
    paths: State<AppPaths>,
    username: String,
    master_password: String,
) -> Result<AuthSuccess, CommandError> {
    do_register(&state, &paths, &username, &master_password)
}

#[tauri::command]
pub fn login(
    state: State<AppState>,
    paths: State<AppPaths>,
    username: String,
    master_password: String,
) -> Result<AuthSuccess, CommandError> {
    do_login(&state, &paths, &username, &master_password)
}

#[tauri::command]
pub fn lock(state: State<AppState>) {
    do_lock(&state);
}

#[tauri::command]
pub fn touch(state: State<AppState>) {
    do_touch(&state);
```

```rust
use crate::state::AppState;
use pwmgr_core::session::Clock;

/// Returns true only on the transition from unlocked to locked (so the caller emits the
/// `vault-locked` event exactly once per lock, not on every tick of an already-locked session).
pub fn tick(state: &AppState, clock: &dyn Clock) -> bool {
    let mut session = state.session.lock().unwrap();
    if session.is_locked() {
        return false;
    }
    session.check_expiry(clock);
    let just_locked = session.is_locked();
    drop(session);

    if just_locked {
        *state.vault.lock().unwrap() = None;
        *state.vault_path.lock().unwrap() = None;
    }
    just_locked
}
```

```ts
// src/hooks/useActivityWatcher.ts
import { useEffect, useRef } from "react";
import { api } from "../lib/api";

const THROTTLE_MS = 5000;

/**
 * Calls the `touch` IPC command (resetting the backend idle timer) on user activity, throttled so
 * a burst of mousemove/keydown events doesn't flood IPC. Only active while `enabled` (i.e. while
 * logged in) - no point touching a session that isn't unlocked.
 */
export function useActivityWatcher(enabled: boolean) {
  const lastSent = useRef(0);

  useEffect(() => {
    if (!enabled) return;

    const onActivity = () => {
      const now = Date.now();
      if (now - lastSent.current < THROTTLE_MS) return;
      lastSent.current = now;
      api.touch().catch(() => {
        // Best-effort - a failed touch just means the next activity event retries. Not worth
        // surfacing to the user; it doesn't affect correctness of the idle lock itself.
      });
    };

    window.addEventListener("mousemove", onActivity);
    window.addEventListener("keydown", onActivity);
    window.addEventListener("mousedown", onActivity);

    return () => {
      window.removeEventListener("mousemove", onActivity);
      window.removeEventListener("keydown", onActivity);
      window.removeEventListener("mousedown", onActivity);
    };
  }, [enabled]);
}
```

```rust
// core/src/vault.rs, inserted between list_credentials_for_site and soft_delete_credential

pub fn update_credential(
    conn: &Connection,
    key: &VaultKey,
    credential_id: i64,
    username: Option<&str>,
    secret: &str,
    remarks: Option<&str>,
) -> Result<(), VaultError> {
    let (secret_enc, nonce) = crate::crypto::seal(key, secret.as_bytes())?;
    let updated = conn.execute(
        "UPDATE credentials SET username = ?1, secret_enc = ?2, secret_nonce = ?3, remarks = ?4,
         updated_at = datetime('now') WHERE id = ?5 AND deleted_at IS NULL",
        rusqlite::params![username, secret_enc, nonce.to_vec(), remarks, credential_id],
    )?;
    if updated == 0 {
        return Err(VaultError::NotFound);
    }
    Ok(())
}
```

**Important — camelCase note:** Tauri auto-converts `#[tauri::command]` *argument* names between
camelCase (JS) and snake_case (Rust) for you, but it does **not** do this for return values —
those serialize via plain `serde_json` using the Rust field names verbatim. Every DTO with a
multi-word field name (`site_id`, `credential_id`, `site_name`, `deleted_at`,
`auto_lock_minutes`, etc.) needs `#[serde(rename_all = "camelCase")]` or its JSON keys will come
out snake_case while the TypeScript interfaces (already written camelCase, matching Tauri's
argument-side convention) expect camelCase — the fields would silently deserialize as `undefined`
on the frontend. Every DTO below and in later tasks (`TrashedCredentialDto` in Task 21,
`ConfigDto` in Task 24) carries this attribute for that reason.

```rust
// src-tauri/src/dto.rs (below AuthSuccess from Task 10)
use pwmgr_core::vault::{Credential, SearchResult, Site};

#[derive(Debug, Serialize, PartialEq)]
#[serde(rename_all = "camelCase")]
pub struct SiteDto {
    pub id: i64,
    pub name: String,
    pub url: Option<String>,
    pub notes: Option<String>,
}

impl From<Site> for SiteDto {
    fn from(s: Site) -> Self {
        SiteDto { id: s.id, name: s.name, url: s.url, notes: s.notes }
    }
}

#[derive(Debug, Serialize, PartialEq)]
#[serde(rename_all = "camelCase")]
pub struct SearchResultDto {
    pub site_id: i64,
    pub site_name: String,
    pub credential_id: i64,
    pub username: Option<String>,
}

impl From<SearchResult> for SearchResultDto {
    fn from(r: SearchResult) -> Self {
        SearchResultDto { site_id: r.site_id, site_name: r.site_name, credential_id: r.credential_id, username: r.username }
    }
}

// No #[derive(Debug)] here on purpose - mirrors pwmgr_core::vault::Credential's manual redacting
// Debug impl (core/src/vault.rs:139) so this DTO can never leak the plaintext secret through a
// stray `{:?}` or `dbg!()` on the Tauri side either.
#[derive(Serialize, PartialEq)]
#[serde(rename_all = "camelCase")]
pub struct CredentialDto {
    pub id: i64,
    pub site_id: i64,
    pub username: Option<String>,
    pub secret: String,
    pub remarks: Option<String>,
}

impl std::fmt::Debug for CredentialDto {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        f.debug_struct("CredentialDto")
            .field("id", &self.id)
            .field("site_id", &self.site_id)
            .field("username", &self.username)
            .field("secret", &"<redacted>")
            .field("remarks", &self.remarks)
            .finish()
    }
}

impl From<Credential> for CredentialDto {
    fn from(c: Credential) -> Self {
        CredentialDto { id: c.id, site_id: c.site_id, username: c.username, secret: c.secret, remarks: c.remarks }
    }
}
```

```tsx
// src/screens/Vault.tsx
import { useEffect, useState } from "react";
import { api, SiteDto, SearchResultDto } from "../lib/api";
import { SiteEditor } from "./SiteEditor";
import { SiteDetail } from "./SiteDetail";

export function Vault({ username }: { username: string }) {
  const [sites, setSites] = useState<SiteDto[]>([]);
  const [query, setQuery] = useState("");
  const [results, setResults] = useState<SearchResultDto[] | null>(null);
  const [selectedSiteId, setSelectedSiteId] = useState<number | null>(null);
  const [addingSite, setAddingSite] = useState(false);

  async function refreshSites() {
    setSites(await api.listSites());
  }

  useEffect(() => {
    refreshSites();
  }, []);

  useEffect(() => {
    if (query.trim() === "") {
      setResults(null);
      return;
    }
    // Real-time search (TRD NFR-3: results update within one UI frame of keystroke) - no debounce,
    // this is a local IPC call against a SQLite index, not a network request.
    api.search(query).then(setResults);
  }, [query]);

  if (selectedSiteId !== null) {
    return (
      <SiteDetail
        site={sites.find((s) => s.id === selectedSiteId)!}
        onBack={() => setSelectedSiteId(null)}
      />
    );
  }

  return (
    <div className="min-h-screen dark:bg-slate-900 dark:text-slate-100 p-4">
      <div className="flex justify-between items-center mb-4">
        <h1 className="text-xl font-semibold">Vault — {username}</h1>
        <div className="space-x-2">
          <button
            onClick={() => setAddingSite(true)}
            className="rounded bg-blue-600 px-3 py-1"
            title="Add a new site"
          >
            Add site
          </button>
          <button
            onClick={() => api.lock()}
            className="rounded bg-slate-700 px-3 py-1"
            title="Lock the vault now"
          >
            Lock
          </button>
        </div>
      </div>

      <input
        className="w-full rounded border px-2 py-1 mb-4 dark:bg-slate-800 dark:border-slate-600"
        placeholder="Search by site, username, or remarks..."
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        title="Search your vault in real time"
      />

      {results !== null ? (
        <ul className="space-y-1">
          {results.map((r) => (
            <li
              key={r.credentialId}
              className="cursor-pointer rounded border px-2 py-1 dark:border-slate-700"
              onClick={() => setSelectedSiteId(r.siteId)}
            >
              {r.siteName} — {r.username ?? "(no username)"}
            </li>
          ))}
          {results.length === 0 && <li className="text-sm opacity-70">No matches.</li>}
        </ul>
      ) : (
        <ul className="space-y-1">
          {sites.map((s) => (
            <li
              key={s.id}
              className="cursor-pointer rounded border px-2 py-1 dark:border-slate-700"

            >
              {s.name}
            </li>
          ))}

        </ul>

          onClose={() => setAddingSite(false)}
```

```rust
// core/src/vault.rs, inserted between restore_credential and the SearchResult/search block

#[derive(Debug, Clone, PartialEq)]
pub struct TrashedCredential {
    pub credential_id: i64,
    pub site_id: i64,
    pub site_name: String,
    pub username: Option<String>,
    pub deleted_at: String,
}

pub fn list_trash(conn: &Connection) -> Result<Vec<TrashedCredential>, VaultError> {
    let mut stmt = conn.prepare(
        "SELECT c.id, c.site_id, s.name, c.username, c.deleted_at
         FROM credentials c
         JOIN sites s ON s.id = c.site_id
         WHERE c.deleted_at IS NOT NULL
         ORDER BY c.deleted_at DESC",
    )?;
    let rows = stmt.query_map([], |row| {
        Ok(TrashedCredential {
            credential_id: row.get(0)?,
            site_id: row.get(1)?,
            site_name: row.get(2)?,
            username: row.get(3)?,
            deleted_at: row.get(4)?,
        })
    })?;
    let mut out = Vec::new();
    for r in rows {
        out.push(r?);
    }
    Ok(out)
}

pub fn purge_trash(conn: &Connection) -> Result<usize, VaultError> {
    let count = conn.execute("DELETE FROM credentials WHERE deleted_at IS NOT NULL", [])?;
    Ok(count)
}
```

```rust
use crate::dto::ConfigDto;

const VALID_AUTO_LOCK_MINUTES: [u32; 3] = [1, 5, 15];

pub fn do_get_config(state: &AppState) -> ConfigDto {
    let config = state.config.lock().unwrap();

pub fn do_update_auto_lock_minutes(
    state: &AppState,
    paths: &AppPaths,
    minutes: u32,
) -> Result<(), CommandError> {
    if !VALID_AUTO_LOCK_MINUTES.contains(&minutes) {
        return Err(CommandError {
            code: "invalid_input".to_string(),
            message: "Auto-lock must be 1, 5, or 15 minutes.".to_string(),
        });
    }
    {
        let mut config = state.config.lock().unwrap();
        config.auto_lock_minutes = minutes;
        config.save(&paths.config_path()).map_err(|_| CommandError {
            code: "internal_error".to_string(),
            message: "Could not save settings.".to_string(),
        })?;
    }
    state
        .session
        .lock()
        .unwrap()
        .set_timeout(std::time::Duration::from_secs(minutes as u64 * 60));
    Ok(())
}

pub fn get_config(state: State<AppState>) -> ConfigDto {
    do_get_config(&state)

pub fn update_auto_lock_minutes(
    state: State<AppState>,

    minutes: u32,
```

```tsx
// src/components/Tooltip.tsx
import { useId, useState, ReactElement, cloneElement, MouseEvent, FocusEvent } from "react";

/**
 * Wraps a single focusable child (a <button> or <select>) and shows a small text tooltip on both
 * mouse hover AND keyboard focus. Deliberate replacement for `title=` across every icon/button —
 * `title` is hover-only in most webviews and doesn't reliably appear on keyboard Tab focus.
 */
export function Tooltip({ text, children }: { text: string; children: ReactElement }) {
  const [visible, setVisible] = useState(false);
  const id = useId();

  const child = cloneElement(children, {
    "aria-describedby": id,
    onMouseEnter: (e: MouseEvent) => {
      setVisible(true);
      (children.props as any).onMouseEnter?.(e);
    },
    onMouseLeave: (e: MouseEvent) => {
      setVisible(false);
      (children.props as any).onMouseLeave?.(e);
    },
    onFocus: (e: FocusEvent) => {
      setVisible(true);
      (children.props as any).onFocus?.(e);
    },
    onBlur: (e: FocusEvent) => {
      setVisible(false);
      (children.props as any).onBlur?.(e);
    },
  } as any);

  return (
    <span className="relative inline-block">
      {child}
      {visible && (
        <span
          id={id}
          role="tooltip"
          className="absolute z-10 -top-8 left-1/2 -translate-x-1/2 whitespace-nowrap rounded bg-slate-950 px-2 py-1 text-xs text-slate-100 shadow"
        >
          {text}
        </span>
      )}
    </span>
  );
}
```
